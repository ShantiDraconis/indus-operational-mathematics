\chapter*{Notation and Conventions}
\addcontentsline{toc}{chapter}{Notation and Conventions}

\section*{Sign References}

Signs are referenced using the standard Mahadevan concordance numbering (M001--M417), with Wells concordance provided as secondary reference where applicable.

\section*{Mathematical Notation}

\begin{itemize}
    \item $\mathbb{N}$: Natural numbers (including zero)
    \item $\mathbb{Z}$: Integers
    \item $\mathcal{S}$: Set of all Indus signs
    \item $\mathcal{O}$: Set of operational signs
    \item $\mathcal{T}$: Set of terminal signs
    \item $\sigma, \tau$: Individual signs
    \item $w = \sigma_1\sigma_2\cdots\sigma_n$: Sign sequence (inscription)
    \item $|w|$: Length of sequence $w$
    \item $f(\sigma)$: Frequency of sign $\sigma$
    \item $P(\sigma|\tau)$: Conditional probability of $\sigma$ following $\tau$
\end{itemize}

\section*{Operational Notation}

\begin{itemize}
    \item $\text{INST}(\sigma)$: Instantiation operator (introduces entity)
    \item $\text{COUNT}(n)$: Counting operator (cardinal value $n$)
    \item $\text{REL}(\sigma, \tau)$: Relational operator (connects $\sigma$ and $\tau$)
    \item $\text{QUAL}(\sigma, q)$: Qualification operator (attributes quality $q$ to $\sigma$)
    \item $\text{CLOSE}(\sigma)$: Closure operator (terminates sequence)
    \item $Q \Rightarrow Q'$: State transition from $Q$ to $Q'$
\end{itemize}

\section*{Automata Notation}

\begin{itemize}
    \item $M = (Q, \Sigma, \delta, q_0, F)$: Finite automaton
    \item $Q$: Set of states
    \item $\Sigma$: Input alphabet
    \item $\delta: Q \times \Sigma \rightarrow Q$: Transition function
    \item $q_0$: Initial state
    \item $F \subseteq Q$: Set of accepting (final) states
\end{itemize}

\section*{Abbreviations}

\begin{itemize}
    \item DFA: Deterministic Finite Automaton
    \item NFA: Non-deterministic Finite Automaton
    \item IVC: Indus Valley Civilization
    \item M-number: Mahadevan concordance number
    \item W-number: Wells concordance number
\end{itemize}
